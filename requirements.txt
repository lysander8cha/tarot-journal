# Python dependencies for the Tarot Journal backend.
#
# Versions are pinned to the exact releases the app was built and
# tested against, so a fresh `pip install -r requirements.txt` in a
# new virtualenv reproduces a known-good environment. When upgrading,
# change the pin deliberately and test the app afterwards.

# Web backend serving the Electron/React frontend. Flask also installs
# Jinja2, which renders the PDF export templates.
Flask==3.1.2

# Image loading/scaling for deck thumbnails.
Pillow==11.3.0

# Astrology charts (chart-integration feature). Kerykeion bundles
# Swiss Ephemeris data; timezonefinder is used to resolve a timezone
# string from stored lat/lon at generation time.
kerykeion==5.12.8
timezonefinder==8.2.0

# Offline place-name lookup (geocoder.py) downloads its dataset with
# requests (bundled certifi avoids macOS urllib certificate issues).
requests==2.32.5

# PDF export of journal entries (renders an HTML/CSS template). Pulls in
# system dependencies (pango, cairo, glib, gdk-pixbuf, libffi) which must
# already be installed at the OS level — see README setup notes.
weasyprint==66.0
# Rasterizes kerykeion's chart SVGs into PNGs before embedding into
# the PDF — WeasyPrint's own inline-SVG renderer mis-renders the
# chart wheel (solid black) because of how kerykeion styles it.
cairosvg==2.8.2

# AI assistant features (Settings → AI, the Scribe). The official
# Anthropic SDK talks to Claude; OpenAI/local providers use requests.
anthropic==0.120.0
# The Scribe reads book files: pypdf extracts PDF text layers, mobi
# unpacks Kindle-format ebooks (EPUB needs only the standard library).
pypdf==6.13.0
mobi==0.4.1

# Development: test suite (run with `npm run test:backend` or
# `.venv/bin/python3 -m pytest`).
pytest==8.4.2

zeroconf>=0.130  # Bonjour/mDNS advertisement for the phone-sync companion

# NOTE: wxPython was removed from this list (2026-07). It powered the
# deprecated legacy Python UI (main.py / mixin_*.py / ui_library/ etc.),
# which the Electron/React frontend replaced and which was deleted from
# the repo in July 2026.
